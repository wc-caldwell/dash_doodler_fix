#############
These are only really needed for when running on macos, since pydensecrf does not exist for macos
You should be able to just install python dependencies and run ```python doodler.py``` on windows machines (and linux)
#############

docker build --platform linux/amd64 -f deploy/Dockerfile -t doodler_docker_image .

mkdir -p app_files/logs

docker run --platform linux/amd64 -p 8050:8050 \
  -v "$(pwd)/app.py:/app.py" \
  -v "$(pwd)/classes.txt:/classes.txt" \
  -v "$(pwd)/files_done.txt:/files_done.txt" \
  -v "$(pwd)/assets:/assets" \
  -v "$(pwd)/results:/results" \
  -v "$(pwd)/labeled:/labeled" \
  -v "$(pwd)/app_files/logs:/app_files/logs" \
  --rm -it doodler_docker_image 2>&1 | tee run.log
